procesamiento.py · limpieza y agregacionesprocesamiento.py · cleaning and aggregations

Lee el CSV con Spark, reconoce encabezados, conserva las columnas relevantes, normaliza texto, convierte precios y fechas, escribe Parquet y prepara resultados de Power BI.

Reads the CSV with Spark, recognizes headers, keeps relevant columns, normalizes text, converts prices and dates, writes Parquet, and prepares Power BI outputs.

# Expresiones regulares para simplificar los nombres de columnas.
import re
# Normalización de acentos y caracteres Unicode.
import unicodedata

# API de DataFrames de Spark y funciones de transformación de columnas.
from pyspark.sql import SparkSession, functions as F
# Rutas y parámetros centralizados del proyecto.
from config import (
    DATOS_ORIGINALES, DATOS_PROCESADOS, MEMORIA_SPARK,
    NUCLEOS_SPARK, COLUMNAS, POWERBI,
)


# Encapsula la lectura, preparación y exportación de datos de contratación.
class ProcesadorSECOP:
        # Inicia una sesión Spark en modo local para procesar el CSV.
    def __init__(self):
        self.spark = (
            SparkSession.builder
            # Nombre que identifica el trabajo dentro de Spark.
            .appName("SECOP-II-BigData")
            # Ejecuta Spark localmente con el número de núcleos configurado.
            .master(f"local[{NUCLEOS_SPARK}]")
            # Memoria asignada al proceso principal de Spark.
            .config("spark.driver.memory", MEMORIA_SPARK)
            # Particiones usadas en intercambios de datos (por ejemplo, agregaciones).
            .config("spark.sql.shuffle.partitions", "8")
            # Zona horaria empleada para interpretar fechas y horas.
            .config("spark.sql.session.timeZone", "America/Bogota")
            .getOrCreate()
        )
        # Reduce la cantidad de mensajes informativos en terminal.
        self.spark.sparkContext.setLogLevel("WARN")

    # Permite utilizar esta función sin crear una instancia del procesador.
    @staticmethod
        # Quita diferencias de acentos, espacios y signos en los encabezados.
    def normalizar_nombre(nombre):
        t = unicodedata.normalize("NFKD", nombre)
        t = "".join(c for c in t if not unicodedata.combining(c))
        return re.sub(r"[^a-z0-9]", "", t.lower())

        # Comprueba que el CSV exista y crea un DataFrame Spark.
    def leer(self):
        if not DATOS_ORIGINALES.is_file():
            raise FileNotFoundError(f"No se encuentra el CSV: {DATOS_ORIGINALES}")
        print("Leyendo SECOP II...")
        return (
            self.spark.read
            # La primera fila contiene los nombres de las columnas.
            .option("header", True)
            # Codificación de los caracteres del archivo.
            .option("encoding", "UTF-8")
            # Admite saltos de línea dentro de campos CSV entrecomillados.
            .option("multiLine", True)
            # Carácter que delimita textos entre comillas.
            .option("quote", '"')
            # Carácter de escape para comillas dentro de los valores.
            .option("escape", '"')
            # Detiene el procesamiento ante filas CSV mal formadas.
            .option("mode", "FAILFAST")
            .csv(str(DATOS_ORIGINALES))
        )

        # Detecta encabezados originales y conserva únicamente los definidos en config.py.
    def seleccionar_columnas(self, df):
        # Asocia la forma normalizada del encabezado con su nombre original.
        disponibles = {self.normalizar_nombre(c): c for c in df.columns}
        expresiones, faltantes = [], []
            # Busca primero el nombre técnico y luego las variantes descriptivas.
        for tecnico, alternativas in COLUMNAS.items():
            original = next(
                (disponibles[self.normalizar_nombre(n)]
                 for n in [tecnico, *alternativas]
                 if self.normalizar_nombre(n) in disponibles),
                None,
            )
            if original is None:
                faltantes.append(tecnico)
            else:
                expresiones.append(F.col(f"`{original}`").alias(tecnico))
        # Informa si el CSV carece de columnas requeridas.
        if faltantes:
            raise ValueError(
                f"Columnas ausentes: {faltantes}. Encabezados leídos: {df.columns}"
            )
        print("Seleccionando columnas...")
        # Selecciona las columnas y les asigna nombres técnicos consistentes.
        return df.select(*expresiones)

        # Prepara textos, valores monetarios y fechas para análisis.
    def limpiar(self, df):
        print("Limpiando información...")
        # Columnas cuyo contenido debe tratarse como texto.
        textos = [
            "id_del_proceso", "entidad", "departamento_entidad",
            "ciudad_entidad", "nombre_del_procedimiento",
            "descripci_n_del_procedimiento", "modalidad_de_contratacion",
            "justificaci_n_modalidad_de", "fase",
        ]
        # Quita espacios sobrantes y deja nulos los textos vacíos.
        for c in textos:
            df = df.withColumn(
                c,
                F.when(F.length(F.trim(F.col(c))) > 0, F.trim(F.col(c)))
            )

        # SECOP publica precio base como valor numérico textual.
        df = df.withColumn(
            "precio_base",
            F.expr("try_cast(regexp_replace(precio_base, '[^0-9.-]', '') AS DOUBLE)")
        )

        # Interpretar las fechas de publicación en distintos formatos.
        # Campo original que contiene la fecha de publicación.
        fecha = F.col("fecha_de_publicacion_del")
        df = df.withColumn(
            "fecha_publicacion",
            # Escoge el primer formato de fecha que se pueda interpretar.
            F.coalesce(
                F.to_date(F.try_to_timestamp(fecha)),
                F.try_to_date(fecha, "yyyy-MM-dd"),
                F.try_to_date(fecha, "dd/MM/yyyy"),
                F.try_to_date(fecha, "MM/dd/yyyy"),
            )
        )
        # Extrae el año para construir tendencias temporales.
        df = df.withColumn("anio", F.year("fecha_publicacion"))
        df = df.withColumn(
            # Une nombre y descripción del procedimiento en un solo texto.
            "texto_pln",
            F.trim(F.regexp_replace(
                F.lower(F.concat_ws(
                    " ", F.col("nombre_del_procedimiento"),
                    F.col("descripci_n_del_procedimiento")
                )), r"\s+", " "
            ))
        )
        # Conserva filas con identificador y al menos 30 caracteres de texto.
        return df.filter(
            F.col("id_del_proceso").isNotNull()
            & (F.length(F.col("texto_pln")) >= 30)
        )

        # Ejecuta la preparación completa y escribe los datos depurados.
    def preparar(self):
        df = self.limpiar(self.seleccionar_columnas(self.leer()))
        # Garantiza que exista el directorio de salida.
        DATOS_PROCESADOS.parent.mkdir(parents=True, exist_ok=True)
        print("Guardando dataset en Parquet... (puede tardar por el volumen)")
        # Sobrescribe la salida previa en formato columnar Parquet.
        df.write.mode("overwrite").parquet(str(DATOS_PROCESADOS))
        print("Dataset procesado correctamente.")

        # Construye tres resúmenes independientes para tableros de Power BI.
    def exportar_powerbi(self):
        df = self.spark.read.parquet(str(DATOS_PROCESADOS))
        POWERBI.mkdir(parents=True, exist_ok=True)
        # Cada clave será una carpeta CSV producida por Spark.
        tablas = {
            # Recuento, suma y promedio del precio base por departamento.
            "contratos_departamento": (
                df.groupBy("departamento_entidad")
                .agg(
                    F.count("*").alias("total_registros"),
                    F.sum("precio_base").alias("precio_base_total"),
                    F.avg("precio_base").alias("precio_base_promedio"),
                )
            ),
            # Recuento y suma del precio base por modalidad contractual.
            "contratos_modalidad": (
                df.groupBy("modalidad_de_contratacion")
                .agg(
                    F.count("*").alias("total_registros"),
                    F.sum("precio_base").alias("precio_base_total"),
                )
            ),
            # Recuento y suma del precio base por año de publicación.
            "contratos_anio": (
                df.groupBy("anio")
                .agg(
                    F.count("*").alias("total_registros"),
                    F.sum("precio_base").alias("precio_base_total"),
                )
            ),
        }
            # Guarda cada resumen como CSV con encabezados.
        for nombre, tabla in tablas.items():
            (tabla.coalesce(1).write.mode("overwrite")
             .option("header", True).csv(str(POWERBI / nombre)))
        print("Tablas agregadas para Power BI generadas.")

        # Libera la sesión Spark y sus recursos.
    def cerrar(self):
        self.spark.stop()

sentimientos.py · clasificación exploratoriasentimientos.py · exploratory classification

Selecciona una muestra limitada, carga el modelo multilingüe y escribe las etiquetas y probabilidades en un CSV.

Selects a bounded sample, loads the multilingual model, and writes labels and probabilities to a CSV.

# Biblioteca estándar para crear archivos CSV de resultados.
import csv
# Motor de inferencia del modelo neuronal en CPU.
import torch
# Interfaz de alto nivel para cargar el clasificador de sentimientos.
from transformers import pipeline
# Spark lee Parquet y obtiene la muestra distribuida.
from pyspark.sql import SparkSession, functions as F
# Parámetros de muestreo, modelo, memoria y rutas.
from config import (
    DATOS_PROCESADOS, ARCHIVO_SENTIMIENTOS, RESULTADOS,
    MODELO_PLN, MAX_REGISTROS_PLN, FRACCION_MUESTRA,
    TAMANO_LOTE, MEMORIA_SPARK, NUCLEOS_SPARK,
)


# Obtiene una muestra y clasifica el sentimiento de sus descripciones.
class AnalizadorSentimientos:
        # El modelo se cargará solo cuando comience la inferencia.
    def __init__(self):
        self.modelo = None

        # Carga un modelo multilingüe y su tokenizador para ejecución local.
    def cargar_modelo(self):
        print("Cargando modelo de sentimientos...")
        self.modelo = pipeline(
            # Tipo de tarea: clasificación de textos.
            task="text-classification",
            # Identificador de los pesos del modelo en Hugging Face.
            model=MODELO_PLN,
            # Tokenizador asociado al mismo modelo.
            tokenizer=MODELO_PLN,
            # -1 indica inferencia en CPU.
            device=-1,
            # Prefiere los pesos en formato Safetensors.
            model_kwargs={"use_safetensors": True},
        )

        # Selecciona textos del Parquet usando Spark sin traer millones de filas.
    def obtener_muestra(self):
        spark = (
            SparkSession.builder.appName("SECOP-Muestra-PLN")
            .master(f"local[{NUCLEOS_SPARK}]")
            .config("spark.driver.memory", MEMORIA_SPARK)
            .getOrCreate()
        )
        spark.sparkContext.setLogLevel("WARN")
        try:
            # Obtiene las variables necesarias para el análisis posterior.
            df = (
                spark.read.parquet(str(DATOS_PROCESADOS))
                .select("id_del_proceso", "entidad", "departamento_entidad",
                        "anio", "precio_base", "texto_pln")
                .filter(F.col("texto_pln").isNotNull())
            )
            # Muestreo sin reemplazo; la semilla permite reproducir la selección.
# limit() restringe el máximo de filas transferidas desde Spark a Python.
            filas = (df.sample(withReplacement=False,
                               fraction=FRACCION_MUESTRA, seed=42)
                     .limit(MAX_REGISTROS_PLN).collect())
            # Convierte las filas Spark recolectadas en diccionarios Python.
            registros = [f.asDict() for f in filas]
            # Cierra Spark, incluso si falla la extracción de la muestra.
        finally:
            spark.stop()
        if not registros:
            raise ValueError("Muestra vacía: aumenta FRACCION_MUESTRA en config.py")
        print(f"Registros seleccionados: {len(registros)}")
        return registros

        # Clasifica cada texto de la muestra y escribe etiquetas y probabilidades.
    def ejecutar(self):
        registros = self.obtener_muestra()
        self.cargar_modelo()
        RESULTADOS.mkdir(parents=True, exist_ok=True)
        # Orden fijo de las columnas del archivo de salida.
        campos = [
            "id_del_proceso", "entidad", "departamento_entidad",
            "anio", "precio_base", "texto_pln", "sentimiento",
            "confianza", "prob_negativo", "prob_neutral", "prob_positivo",
        ]
        # Traducción de las tres etiquetas del modelo al español.
        etiquetas = {"negative": "Negativo", "neutral": "Neutral", "positive": "Positivo"}
        # UTF-8 con BOM facilita la lectura de tildes en aplicaciones Windows.
        with open(ARCHIVO_SENTIMIENTOS, "w", encoding="utf-8-sig", newline="") as archivo:
            escritor = csv.DictWriter(archivo, fieldnames=campos)
            escritor.writeheader()
            # Procesamiento por lotes para controlar el uso de memoria.
            for inicio in range(0, len(registros), TAMANO_LOTE):
                lote = registros[inicio:inicio + TAMANO_LOTE]
                # Extrae únicamente los textos del lote actual.
                textos = [r["texto_pln"] for r in lote]
                # Desactiva cálculos de gradientes durante la predicción.
                with torch.inference_mode():
                    predicciones = self.modelo(
                        # Recorta las entradas largas para limitar tokens y memoria.
                        textos, truncation=True, max_length=256,
                        # Recupera puntuaciones de todas las clases, no solo la ganadora.
                        top_k=None, batch_size=TAMANO_LOTE,
                    )
                # Une cada predicción con el registro al que pertenece.
                for registro, prediccion in zip(lote, predicciones):
                    # Convierte las puntuaciones del modelo en un diccionario.
                    probs = {x["label"].lower(): float(x["score"]) for x in prediccion}
                    # Selecciona la clase con mayor puntuación.
                    mejor = max(probs, key=probs.get)
                    # Guarda el registro original y las tres probabilidades.
                    escritor.writerow({
                        **registro,
                        "sentimiento": etiquetas.get(mejor, mejor),
                        "confianza": probs[mejor],
                        "prob_negativo": probs.get("negative", 0),
                        "prob_neutral": probs.get("neutral", 0),
                        "prob_positivo": probs.get("positive", 0),
                    })
                print(f"Procesados {min(inicio + TAMANO_LOTE, len(registros))}/{len(registros)}")
        print(f"Resultados: {ARCHIVO_SENTIMIENTOS}")

main.py · orquestaciónmain.py · orchestration

Permite ejecutar ingeniería, PLN, análisis o todo el flujo desde un único punto de entrada.

Runs engineering, NLP, analysis, or the complete workflow from one entry point.

# Permite seleccionar la etapa de ejecución desde la terminal.
import argparse
# Clase responsable de preparar datos y generar tablas agregadas.
from procesamiento import ProcesadorSECOP
# Clase responsable de ejecutar el modelo de sentimientos.
from sentimientos import AnalizadorSentimientos


# Etapa de ingeniería de datos: CSV original -> Parquet.
def preparar_datos():
    # Crea una sesión Spark para la operación solicitada.
    p = ProcesadorSECOP()
    try:
        p.preparar()
    # Garantiza la liberación de recursos aunque ocurra un error.
    finally:
        p.cerrar()


# Etapa de PLN: selección de muestra y clasificación de textos.
def analizar_sentimientos():
    AnalizadorSentimientos().ejecutar()


# Etapa analítica: agrupa registros y genera archivos para Power BI.
def generar_reportes():
    # Crea una sesión Spark para la operación solicitada.
    p = ProcesadorSECOP()
    try:
        p.exportar_powerbi()
    # Garantiza la liberación de recursos aunque ocurra un error.
    finally:
        p.cerrar()


# Ejecuta las tres etapas en el orden de dependencia.
def ejecutar_todo():
    preparar_datos()
    analizar_sentimientos()
    generar_reportes()


# Interpreta el argumento de línea de comandos y ejecuta la acción.
def main():
    parser = argparse.ArgumentParser(description="SECOP II: Spark, PLN y Power BI")
    # Acciones disponibles: preparar, sentimientos, reportes y todo.
    parser.add_argument("accion", choices=["preparar", "sentimientos", "reportes", "todo"])
    args = parser.parse_args()
    # Relaciona cada nombre de acción con su función Python.
    acciones = {
        "preparar": preparar_datos,
        "sentimientos": analizar_sentimientos,
        "reportes": generar_reportes,
        "todo": ejecutar_todo,
    }
    # Llama únicamente la función solicitada por el usuario.
    acciones[args.accion]()


# Evita ejecutar el programa automáticamente cuando se importa como módulo.
if __name__ == "__main__":
    main()
